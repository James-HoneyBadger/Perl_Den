=head2 The Perl One-Liner Mindset

Perl shines at quick text transformations.  These patterns scale from

=head2 Reading Files — Multiple Approaches

    # Line by line (memory efficient)
    open my $fh, '<', $file or die $!;
    while (my $line = <$fh>) {
        chomp $line;
        process($line);
    }
    close $fh;
    
    # Slurp entire file
    open my $fh, '<', $file or die $!;
    local $/;
    my $content = <$fh>;
    close $fh;
    
    # Into an array
    open my $fh, '<', $file or die $!;
    my @lines = <$fh>;
    close $fh;
    chomp @lines;

=head2 CSV Processing

    # Simple (no quoting)
    while (<$fh>) {
        chomp;
        my @fields = split /,/;
        printf "Name: %s, Age: %s\n", $fields[0], $fields[1];
    }
    
    # With Text::CSV (handles quoting, escaping)
    use Text::CSV;
    my $csv = Text::CSV->new({ binary => 1, auto_diag => 1 });
    
    open my $fh, '<:encoding(UTF-8)', 'data.csv' or die $!;
    my $header = $csv->getline($fh);
    $csv->column_names(@$header);
    
    while (my $row = $csv->getline_hr($fh)) {
        printf "%s (age %d)\n", $row->{name}, $row->{age};
    }
    close $fh;

=head2 Config File Parsing

    # INI-style (simple)
    my %config;
    my $section = 'default';
    open my $fh, '<', '/etc/myapp.conf' or die $!;
    while (<$fh>) {
        chomp;
        s/#.*//;               # strip comments
        s/^\s+|\s+$//g;        # trim
        next unless length;
        
        if (/^\[(.+)\]$/) {
            $section = $1;
        } elsif (/^(\w+)\s*=\s*(.*)/) {
            $config{$section}{$1} = $2;
        }
    }
    close $fh;
    
    # YAML
    use YAML::XS qw(LoadFile);
    my $config = LoadFile('/etc/myapp.yml');
    
    # JSON
    use JSON::MaybeXS;
    open my $jfh, '<', 'config.json' or die $!;
    my $data = decode_json(do { local $/; <$jfh> });
    close $jfh;

    my $re = qr{

=head2 Multi-File Search and Replace

    use File::Find;
    
    my $search  = qr/deprecated_function/;
    my $replace = 'new_function';
    my $count   = 0;
    
    find(sub {
        return unless /\.pl$|\.pm$/;
        
        open my $fh, '<', $_ or return;
        my $content = do { local $/; <$fh> };
        close $fh;
        
        if ($content =~ s/$search/$replace/g) {
            open my $out, '>', $_ or die $!;
            print $out $content;
            close $out;
            $count++;
            print "Updated: $File::Find::name\n";
        }
    }, 'lib/');
    
    print "Modified $count files\n";

=head2 Generating Reports

    # Columnar output
    my @data = (
        ['Host',  'Status', 'Uptime'],
        ['web-01', 'UP',    '45d'],
        ['web-02', 'DOWN',  '0d'],
        ['db-01',  'UP',    '120d'],
    );
    
    # Calculate column widths
    my @widths;
    for my $row (@data) {
        for my $i (0..$#$row) {
            my $len = length($row->[$i]);
            $widths[$i] = $len if !$widths[$i] || $len > $widths[$i];
        }
    }
    
    # Print with formatting
    my $fmt = join('  ', map { "%-${_}s" } @widths) . "\n";
    printf $fmt, @{$data[0]};
    print '-' x 40, "\n";
    for my $row (@data[1..$#data]) {
        printf $fmt, @$row;
    }
