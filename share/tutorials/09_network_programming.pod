=head1 Network Programming with Perl

=head2 Overview

Perl provides excellent networking capabilities, from simple HTTP
requests to raw socket programming and DNS queries.

=head2 HTTP Requests with HTTP::Tiny

HTTP::Tiny is included in Perl core since 5.14:

    use HTTP::Tiny;
    
    my $http = HTTP::Tiny->new(timeout => 30);
    
    # GET
    my $resp = $http->get('https://api.example.com/status');
    if ($resp->{success}) {
        print $resp->{content};
    } else {
        warn "Failed: $resp->{status} $resp->{reason}\n";
    }
    
    # POST with JSON
    use JSON::MaybeXS qw(encode_json decode_json);
    
    my $resp = $http->post('https://api.example.com/data', {
        content => encode_json({ key => 'value' }),
        headers => { 'Content-Type' => 'application/json' },
    });
    
    my $data = decode_json($resp->{content});

=head2 Socket Programming

    use IO::Socket::INET;
    
    # TCP client
    my $sock = IO::Socket::INET->new(
        PeerAddr => 'example.com',
        PeerPort => 80,
        Proto    => 'tcp',
        Timeout  => 10,
    ) or die "Cannot connect: $!\n";
    
    print $sock "GET / HTTP/1.0\r\nHost: example.com\r\n\r\n";
    while (<$sock>) { print }
    close $sock;
    
    # TCP server
    my $server = IO::Socket::INET->new(
        LocalPort => 8080,
        Proto     => 'tcp',
        Listen    => 5,
        Reuse     => 1,
    ) or die "Cannot listen: $!\n";
    
    print "Listening on port 8080...\n";
    while (my $client = $server->accept()) {
        my $peer = $client->peerhost();
        print "Connection from $peer\n";
        print $client "Hello from Perl server!\n";
        close $client;
    }

=head2 SSL/TLS Connections

    use IO::Socket::SSL;
    
    my $ssl = IO::Socket::SSL->new(
        PeerAddr => 'example.com',
        PeerPort => 443,
        SSL_verify_mode => IO::Socket::SSL::SSL_VERIFY_PEER(),
    ) or die "SSL connection failed: $!\n";
    
    print $ssl "GET / HTTP/1.1\r\nHost: example.com\r\n\r\n";
    
    # Check certificate
    my $subject = $ssl->peer_certificate('subject');
    print "Certificate: $subject\n";
    
    close $ssl;

=head2 DNS Queries

    use Net::DNS;
    
    my $resolver = Net::DNS::Resolver->new;
    
    # A record
    my $reply = $resolver->query('example.com', 'A');
    if ($reply) {
        for my $rr ($reply->answer) {
            next unless $rr->type eq 'A';
            print "IP: " . $rr->address . "\n";
        }
    }
    
    # MX records
    my $mx_reply = $resolver->query('example.com', 'MX');
    if ($mx_reply) {
        for my $rr (sort { $a->preference <=> $b->preference }
                    grep { $_->type eq 'MX' } $mx_reply->answer) {
            printf "MX: %d %s\n", $rr->preference, $rr->exchange;
        }
    }
    
    # Reverse DNS
    my $rev = $resolver->query('4.4.8.8.in-addr.arpa', 'PTR');

=head2 Port Scanning

    use IO::Socket::INET;
    
    my $host = '192.168.1.1';
    my @common_ports = (21, 22, 23, 25, 53, 80, 110, 143, 443, 993, 995,
                        3306, 5432, 6379, 8080, 8443);
    
    for my $port (@common_ports) {
        my $sock = IO::Socket::INET->new(
            PeerAddr => $host,
            PeerPort => $port,
            Proto    => 'tcp',
            Timeout  => 1,
        );
        if ($sock) {
            printf "Port %-5d OPEN\n", $port;
            close $sock;
        }
    }

=head2 Monitoring with Ping

    # Using system ping
    sub ping_host {
        my ($host) = @_;
        my $output = `ping -c 1 -W 2 $host 2>&1`;
        if ($output =~ /time=([\d.]+)\s*ms/) {
            return { alive => 1, rtt => $1 };
        }
        return { alive => 0 };
    }
    
    my @servers = ('web-01', 'db-01', 'app-01');
    for my $host (@servers) {
        my $r = ping_host($host);
        printf "%-15s %s\n", $host,
            $r->{alive} ? "UP (${$r}{rtt}ms)" : "DOWN";
    }

=head2 Downloading Files

    use HTTP::Tiny;
    
    my $http = HTTP::Tiny->new;
    my $url  = 'https://example.com/data.tar.gz';
    my $file = '/tmp/data.tar.gz';
    
    my $resp = $http->mirror($url, $file);
    if ($resp->{success}) {
        print "Downloaded to $file\n";
    } else {
        die "Download failed: $resp->{status}\n";
    }

=head2 Sending Notifications

    # Simple email via sendmail
    sub send_alert {
        my ($to, $subject, $body) = @_;
        open my $mail, '|-', '/usr/sbin/sendmail', '-t'
            or die "Cannot send mail: $!\n";
        print $mail "To: $to\n";
        print $mail "Subject: $subject\n";
        print $mail "\n";
        print $mail $body;
        close $mail;
    }
    
    send_alert(
        'admin@example.com',
        'ALERT: Disk Space Low',
        "Server web-01 is at 95% disk usage.\n"
    );

=head2 Practical: Health Check Script

    #!/usr/bin/env perl
    use strict;
    use warnings;
    use HTTP::Tiny;
    use IO::Socket::INET;
    
    my @checks = (
        { name => 'Web',  host => 'localhost', port => 80,   path => '/' },
        { name => 'API',  host => 'localhost', port => 8080, path => '/health' },
        { name => 'DB',   host => 'localhost', port => 5432 },
        { name => 'SSH',  host => 'localhost', port => 22 },
    );
    
    my $http = HTTP::Tiny->new(timeout => 5);
    
    for my $check (@checks) {
        my $status;
        if ($check->{path}) {
            my $url = "http://$check->{host}:$check->{port}$check->{path}";
            my $r = $http->get($url);
            $status = $r->{success} ? "OK ($r->{status})" : "FAIL ($r->{status})";
        } else {
            my $sock = IO::Socket::INET->new(
                PeerAddr => $check->{host},
                PeerPort => $check->{port},
                Proto    => 'tcp',
                Timeout  => 3,
            );
            $status = $sock ? 'OK' : 'FAIL';
            close $sock if $sock;
        }
        printf "%-10s %s:%d  %s\n", $check->{name},
            $check->{host}, $check->{port}, $status;
    }

=cut
